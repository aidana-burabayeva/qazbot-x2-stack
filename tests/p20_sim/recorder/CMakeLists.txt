cmake_minimum_required(VERSION 3.16)
project(p20_timing_recorder)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(aimdk_msgs REQUIRED)
add_executable(timing_recorder timing_recorder.cpp)
target_compile_features(timing_recorder PRIVATE cxx_std_17)
target_compile_options(timing_recorder PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(timing_recorder rclcpp nav_msgs geometry_msgs std_msgs aimdk_msgs)
install(TARGETS timing_recorder DESTINATION lib/${PROJECT_NAME})
ament_package()
