cmake_minimum_required(VERSION 3.8)
project(qazbot_pose_refine)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep nav2_core nav2_regulated_pure_pursuit_controller nav2_costmap_2d nav2_util pluginlib rclcpp rclcpp_lifecycle tf2_ros tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(qazbot_pose_refine SHARED controller.cpp)
ament_target_dependencies(qazbot_pose_refine nav2_core nav2_regulated_pure_pursuit_controller nav2_costmap_2d nav2_util pluginlib rclcpp rclcpp_lifecycle tf2_ros tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS qazbot_pose_refine DESTINATION lib)
ament_package()
