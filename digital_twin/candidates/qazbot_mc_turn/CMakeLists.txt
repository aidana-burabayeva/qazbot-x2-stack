cmake_minimum_required(VERSION 3.8)
project(qazbot_mc_turn)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(qazbot_mc_turn SHARED src/controllers.cpp)
ament_target_dependencies(qazbot_mc_turn nav2_core nav2_util nav2_controller nav2_regulated_pure_pursuit_controller pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS qazbot_mc_turn LIBRARY DESTINATION lib)
ament_package()
