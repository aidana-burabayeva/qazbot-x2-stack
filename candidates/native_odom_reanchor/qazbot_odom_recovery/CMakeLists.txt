cmake_minimum_required(VERSION 3.16)
project(qazbot_odom_recovery)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(aimdk_msgs REQUIRED)
find_package(qazbot_odom_recovery_msgs REQUIRED)
add_executable(recovery_node src/recovery_node.cpp)
target_compile_features(recovery_node PRIVATE cxx_std_17)
target_compile_options(recovery_node PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(recovery_node rclcpp nav_msgs diagnostic_msgs aimdk_msgs qazbot_odom_recovery_msgs)
add_executable(replay_source src/replay_source.cpp)
target_compile_features(replay_source PRIVATE cxx_std_17)
target_compile_options(replay_source PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(replay_source rclcpp nav_msgs aimdk_msgs)
install(TARGETS replay_source recovery_node DESTINATION lib/${PROJECT_NAME})
ament_package()
