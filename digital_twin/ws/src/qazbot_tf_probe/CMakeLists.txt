cmake_minimum_required(VERSION 3.16)
project(qazbot_tf_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_executable(tf_owner_probe src/tf_owner_probe.cpp)
target_compile_features(tf_owner_probe PRIVATE cxx_std_17)
target_compile_options(tf_owner_probe PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(tf_owner_probe rclcpp tf2_msgs geometry_msgs)
install(TARGETS tf_owner_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
