"""Upstream Nav2 for QORGAN, in the qorgan_map frame, under its own namespace.

Runs the nav2_* binaries from /opt/ros/humble (1.1.20) — nothing is built from
source. Localization is ours (pose_tracker); the map is served latched by
qorgan_map_server directly in qorgan_map, bypassing the vendor map_server.

WHY THE NAMESPACE
-----------------
The vendor navigo stack is autostarted by robot-launch and is ALREADY running
nodes called bt_navigator, controller_server, planner_server, behavior_server,
global_costmap/global_costmap and local_costmap/local_costmap, and already owns
the action names /navigate_to_pose, /compute_path_to_pose, /follow_path (as
robots_dog_msgs types) plus /spin, /wait, /drive_on_heading (as nav2_msgs).

Launching stock Nav2 at the default names would collide with every one of them.
The worst case is lifecycle: our lifecycle_manager calls
`<node>/change_state`, which would land on the VENDOR's servers and can
deactivate live navigation on a robot that physically moves.

So the whole Nav2 stack is pushed under `namespace` (default `qorgan_nav2`):

    /qorgan_nav2/bt_navigator, /qorgan_nav2/controller_server, ...
    /qorgan_nav2/navigate_to_pose, /qorgan_nav2/plan, ...
    /qorgan_nav2/controller_server/change_state

Nothing of the vendor's is touched, and the two stacks can run side by side.

TF IS DELIBERATELY *NOT* NAMESPACED. nav2_bringup's multi-robot launch remaps
('/tf','tf') so each robot gets a private tree — we want the opposite: one
shared tree with the vendor's odom->base_link and our qorgan_map->map. So that
remapping is intentionally absent here.

pose_tracker and qorgan_map_server stay OUTSIDE the namespace: their topics
(/qorgan/localization/pose, /qorgan/map) are consumed by our other tools. If
your localization.launch.py is already running them, pass use_localizer:=false
and/or use_map_server:=false or you will get duplicate node names.

FRAME CHAIN
-----------
    qorgan_map --(pose_tracker, 50 Hz)--> map --(vendor identity, /tf_static)-->
    odom --(robot_tf)--> base_link --(our static publisher)--> laser

pose_tracker publishes `map_frame -> odom_frame`. We set odom_frame:=map so it
emits `qorgan_map -> map`, inserting ABOVE the vendor's un-hijackable identity
instead of fighting it for `odom`'s parent slot.

Velocity goes to /qorgan/cmd_vel (absolute), NOT /cmd_vel: the vendor's
velocity_optimizer owns /cmd_vel and only one publisher may exist there.

    ros2 launch qorgan_nav2 nav2.launch.py map_yaml:=/home/firefly/data/maps/office4.yaml
    ros2 action send_goal /qorgan_nav2/navigate_to_pose \
        nav2_msgs/action/NavigateToPose "{pose: {header: {frame_id: qorgan_map}, ...}}"
"""

import os

from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import (DeclareLaunchArgument, GroupAction,
                            OpaqueFunction)
from launch.conditions import IfCondition
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node, PushRosNamespace
from nav2_common.launch import RewrittenYaml


def _launch_setup(context, *args, **kwargs):
    # `map_yaml` wins when set (bringup passes the path it already resolved);
    # otherwise `map` is a registry name. See map_registry.
    map_yaml = LaunchConfiguration('map_yaml').perform(context).strip()
    if not map_yaml:
        from qorgan_nav2 import map_registry
        name = LaunchConfiguration('map').perform(context).strip()
        entry = map_registry.resolve(name or map_registry.default_map_name())
        map_yaml = entry.map_yaml
        print(f'[nav2] map {entry.name}  {map_yaml}')

    namespace = LaunchConfiguration('namespace')
    params_file = LaunchConfiguration('params_file')
    cmd_vel_topic = LaunchConfiguration('cmd_vel_topic')
    use_localizer = LaunchConfiguration('use_localizer')
    use_map_server = LaunchConfiguration('use_map_server')
    use_zone_server = LaunchConfiguration('use_zone_server')
    autostart = LaunchConfiguration('autostart')

    return _nodes(map_yaml, namespace, params_file, cmd_vel_topic,
                  use_localizer, use_map_server, use_zone_server, autostart)


def _args():
    pkg = get_package_share_directory('qorgan_nav2')
    default_params = os.path.join(pkg, 'config', 'nav2_params.yaml')
    return [
        DeclareLaunchArgument(
            'namespace', default_value='qorgan_nav2',
            description='Namespace for the whole Nav2 stack. Keeps every node, '
                        'action and lifecycle service clear of the vendor navigo '
                        'stack, which owns the un-namespaced names.'),
        DeclareLaunchArgument(
            'map', default_value='',
            description='Map name from the registry (see map_registry). '
                        'Empty = the `default:` in index.yaml.'),
        DeclareLaunchArgument(
            'map_yaml', default_value='',
            description='Explicit map path, overriding `map`. bringup passes '
                        'the path it already resolved through this.'),
        DeclareLaunchArgument(
            'params_file', default_value=default_params,
            description='Nav2 parameters.'),
        DeclareLaunchArgument(
            'cmd_vel_topic', default_value='/qorgan/cmd_vel',
            description='Velocity out (absolute). Do NOT set /cmd_vel while the '
                        'vendor stack runs — velocity_optimizer already owns it.'),
        DeclareLaunchArgument(
            'use_localizer', default_value='true',
            description='Start our pose_tracker (qorgan_map->map). Set false if '
                        'localization.launch.py already runs it.'),
        DeclareLaunchArgument(
            'use_map_server', default_value='true',
            description='Start qorgan_map_server. Set false if it already runs.'),
        DeclareLaunchArgument(
            'use_zone_server', default_value='true',
            description='Start zone_server, which serves the keepout mask '
                        'both costmaps filter on. Set false if it already '
                        'runs. With it off the `keepout_filter` entries in '
                        'nav2_params.yaml simply never receive a mask and '
                        'no zone is enforced.'),
        DeclareLaunchArgument(
            'autostart', default_value='true',
            description='Lifecycle-manage the Nav2 nodes up to ACTIVE.'),
    ]


def _nodes(map_yaml, namespace, params_file, cmd_vel_topic,
           use_localizer, use_map_server, use_zone_server, autostart):
    # Wrap the params under the namespace so /qorgan_nav2/controller_server picks
    # up the `controller_server:` block. Same mechanism nav2_bringup uses.
    configured_params = RewrittenYaml(
        source_file=params_file,
        root_key=namespace,
        param_rewrites={},
        convert_types=True)

    # --- ours: map + localization (NOT namespaced — other tools read these) ---
    map_server = Node(
        condition=IfCondition(use_map_server),
        package='qorgan_nav2', executable='qorgan_map_server',
        name='qorgan_map_server', output='screen',
        parameters=[{'map_yaml': map_yaml,
                     'map_frame': 'qorgan_map',
                     'topic': '/qorgan/map'}],
    )

    # Keepout zones. Deliberately NOT under the lifecycle manager: staying a
    # plain node is what lets it republish a mask the moment an operator
    # draws one, with no transition and no restart.
    zone_server = Node(
        condition=IfCondition(use_zone_server),
        package='qorgan_nav2', executable='zone_server',
        name='zone_server', output='screen',
        parameters=[{'map_yaml': map_yaml,
                     'map_frame': 'qorgan_map'}],
    )

    # odom_frame:=map is the whole trick — see module docstring.
    pose_tracker = Node(
        condition=IfCondition(use_localizer),
        package='qorgan_nav2', executable='pose_tracker',
        name='pose_tracker', output='screen',
        parameters=[{'map_yaml': map_yaml,
                     'map_frame': 'qorgan_map',
                     'odom_frame': 'map',
                     'base_frame': 'base_link',
                     'publish_tf': True,
                     # Leave the vendor's /localization_state alone: publishing
                     # status=3 starts their PATROL and changes a live robot.
                     'publish_localization_state': False}],
    )

    # --- upstream Nav2, namespaced (binaries from /opt/ros/humble) -----------
    # NOTE: no ('/tf','tf') remapping — TF must stay global and shared.
    remaps = [('cmd_vel', cmd_vel_topic)]

    nav2 = GroupAction([
        PushRosNamespace(namespace),
        Node(package='nav2_controller', executable='controller_server',
             name='controller_server', output='screen',
             parameters=[configured_params], remappings=remaps),
        Node(package='nav2_smoother', executable='smoother_server',
             name='smoother_server', output='screen',
             parameters=[configured_params], remappings=remaps),
        Node(package='nav2_planner', executable='planner_server',
             name='planner_server', output='screen',
             parameters=[configured_params], remappings=remaps),
        Node(package='nav2_behaviors', executable='behavior_server',
             name='behavior_server', output='screen',
             parameters=[configured_params], remappings=remaps),
        Node(package='nav2_bt_navigator', executable='bt_navigator',
             name='bt_navigator', output='screen',
             parameters=[configured_params], remappings=remaps),
        Node(package='nav2_lifecycle_manager', executable='lifecycle_manager',
             name='lifecycle_manager_navigation', output='screen',
             parameters=[configured_params, {'autostart': autostart}]),
    ])

    return [map_server, zone_server, pose_tracker, nav2]


def generate_launch_description():
    return LaunchDescription(_args() + [OpaqueFunction(function=_launch_setup)])
# Upstream Nav2 (1.1.20, /opt/ros/humble) on the QORGAN D1 Ultra-W.
#
# Where the numbers come from:
#   * PHYSICAL + TUNING (footprint, resolution, accel, ranges, costmap sizes,
#     tolerances) — the vendor's own tuned file, XG_nav2_params.yaml, from the
#     audit at
#       agibot_slam_nav2/robot_navigation_audit_firefly_20260902_105214/
#       robot_navigation_audit/config/
#         opt_robot_robot_nav_install_robot_nav2_share_robot_nav2_params_XG_nav2_params.yaml
#     XG is OUR robot, not ZG: the live parameter dump the audit captured from
#     the running controller_server (audit .../parameters/controller_server.yaml)
#     matches XG exactly — robot_length 0.65, robot_width 0.5, footprint_points
#     [0.25,0.15,...], max_angular_accel 7.0, lookahead_distance 0.85 — while ZG
#     describes a 1.2 x 0.6 m machine. The vendor bringup also defaults
#     robot_type:='xg', and /opt/export/config/xg_wheel-user-parameters.yaml
#     confirms xg == the wheeled variant.
#   * STRUCTURE — upstream Nav2 plugins. The vendor file's ~97 `navigo_*` plugin
#     classes CANNOT be loaded here: navigo is a fork with its own pluginlib
#     base types. Only `nav2_smoother::SimpleSmoother` is literally shared, so
#     every vendor VALUE below is mapped onto the nearest upstream plugin.
#   * REMAPPED BY US — frames (qorgan_map), map topic (/qorgan/map), cmd_vel
#     (/qorgan/cmd_vel, set in nav2.launch.py), and the /qorgan_nav2 namespace.
#
# Frame plan (see slam_nav2_structure.md / vendor_nav_frames.yaml):
#   qorgan_map --(pose_tracker correction, 50 Hz)--> map --(vendor identity)-->
#   odom --(robot_tf)--> base_link --(static)--> laser
# We insert ABOVE `map` because robot_tf pins an identity map->odom on
# /tf_static at 50 Hz and tf2 keys its cache by CHILD frame — that edge is
# un-hijackable. Nobody writes `map` as a child, so that is where we attach.
#
# AMCL is deliberately absent: our pose_tracker is the localizer.
#
# NOTE: CMakeLists installs config/ by COPY. Editing this file changes nothing
# until you rebuild:
#     colcon build --symlink-install --packages-select qorgan_nav2
# With --symlink-install the installed YAML is a symlink back here and later
# edits apply on the next launch with no rebuild.
#
# ---------------------------------------------------------------------------
# WHY THIS FILE CHANGED: "cannot create a path when a human blocks the robot"
# ---------------------------------------------------------------------------
# Four independent defects conspired to make the planner return no path once a
# person stood close, and to keep the "shell" around long after they moved:
# Historical account below: the current retained footprint is 0.50 x 0.40 m,
# not the vendor's 0.50 x 0.30 m polygon mentioned in defect 1.
#
#   1. FOOTPRINT WAS TOO BIG. We had 0.65 x 0.40, taken from the vendor's
#      robot_length/robot_width (the physical bounding box). The polygon the
#      vendor actually PLANS with is footprint_points = 0.50 x 0.30. This
#      matters more than it looks: nav2_costmap_2d's InflationLayer paints
#      INSCRIBED_INFLATED_OBSTACLE (253) out to the *inscribed radius of the
#      footprint*, and NavFn promotes every cell >= 253 to a hard obstacle
#      (NavFn::setCostmap). So each person was ringed by a 0.20 m no-go band
#      instead of 0.15 m. Two such bands (person + wall) merge across a
#      corridor and the corridor is sealed -> "Failed to create plan".
#      NOTE this is also why inflation_radius is NOT the culprit and is left
#      generous below: inflation_radius only controls the <=252 soft gradient,
#      which NavFn happily drives through. Only the inscribed band blocks.
#
#   2. allow_unknown WAS false WHILE track_unknown_space IS true. Every cell
#      the SLAM map never saw is NO_INFORMATION (255) = hard obstacle. The
#      detour around a person almost always clips such a cell (map edges, the
#      strip behind furniture), so the one available path was rejected. The
#      vendor does not track unknown at all on their global costmap and sets
#      track_unknown_space: False on their behavior_server. We keep tracking it
#      (RViz stays honest, and the controller still refuses real obstacles) but
#      the planner may now route through it.
#
#   3. GHOST OBSTACLES NEVER CLEARED — this is the "shell that stays after the
#      human walks away". Two missing keys, both set by the vendor:
#        * inf_is_valid was unset (false). RPLIDAR reports `inf` for a ray that
#          hits nothing. With inf_is_valid false those rays are DISCARDED, so
#          the ray that used to hit the person and now returns inf never
#          raytrace-clears the old mark. The person is painted into the costmap
#          permanently. Vendor: inf_is_valid: True.
#        * raytrace_min_range was 0.35, so nothing closer than 0.35 m could
#          EVER be cleared, only marked. Vendor uses raytrace_min_range: 0.0
#          (clear from zero) with a non-zero obstacle_min_range (mark only
#          beyond the sensor's valid minimum). We now do the same.
#      A DenoiseLayer is added on top to drop isolated speckle returns before
#      they get inflated into a shell.
#
#   4. THE CONTROLLER COULD NOT DEVIATE. RegulatedPurePursuit is a path
#      FOLLOWER: it tracks the global path and slows/stops on collision, it
#      never generates an alternative. The vendor's controller is not a
#      follower — navigo_local_planner_controller::LocalPlannerController is a
#      sampling LOCAL PLANNER (path_scale / path_range_step /
#      max_rotation_samples / search_radius / base_candidate_plan). The
#      upstream equivalent is DWB, so FollowPath is now DWB behind a rotation
#      shim. The old RPP block is preserved verbatim as FollowPathRPP — see
#      controller_plugins for the one-line revert.
#
# Also: the global obstacle layer now has footprint_clearing_enabled, so the
# robot's own body is always carved free and it can never be trapped inside a
# shell that the planner then refuses to expand out of.
#
# Deliberately NOT copied from the vendor:
#   * Their global costmap has NO obstacle_layer at all (static + electronic
#     map + inflation + frontier only), so a person never reaches their global
#     planner. That works for them because their local planner re-plans around
#     obstacles by itself. With upstream Nav2 we keep the obstacle layer — it
#     is what lets the GLOBAL planner route around a person standing in a
#     doorway — and instead make it forgetful (defect 3) and non-trapping.
#   * inflation_radius 0.05/0.1. Their planner does explicit corridor clearance
#     (corridor_builder: robot_radius 0.35, safety_margin 0.15) so it needs no
#     inflation gradient. NavFn does, so we use 1.20 while preserving the
#     full 0.50 x 0.40 collision footprint through narrow passages.
#   * navigo_costmap_2d::ESDFLayer / FrontierLayer / ElectronicMapLayer, the
#     spatial-temporal planner, MPPI/SocialMPC forks — no upstream equivalent
#     that shares their parameters.

controller_server:

    use_sim_time: false
    # /laser_scan is 7.7 Hz. Running the controller much faster than the sensor
    # just recomputes on stale obstacle data. (Vendor runs 50 Hz, but their
    # controller fuses odom between scans; DWB does not.)
    controller_frequency: 10.0
    odom_topic: /odom/current_pose
    min_x_velocity_threshold: 0.001
    min_y_velocity_threshold: 0.5       # differential drive: no lateral motion
    min_theta_velocity_threshold: 0.001
    # One missed DWB cycle must not trigger recovery while the 1 Hz global
    # replan is still reacting to a newly seen obstacle. This is a bounded
    # grace period, not permission to accept a collision trajectory.
    failure_tolerance: 1.5
    progress_checker_plugin: "progress_checker"
    goal_checker_plugins: ["general_goal_checker"]

    # FollowPath is DWB behind a rotation shim — see defect 4 in the header.
    # To revert to the previous pure-pursuit behaviour, put FollowPathRPP
    # first in this list (nav2_controller uses controller_plugins[0] when the
    # BT passes an empty controller_id) and relaunch. Both stay loaded, so a
    # ControllerSelector BT node can also switch them at runtime.
    controller_plugins: ["FollowPath", "FollowPathRPP"]

    progress_checker:
      plugin: "nav2_controller::SimpleProgressChecker"
      # A careful doorway crossing can make real progress without moving half
      # a metre in one window. Keep the 20 s timeout for a genuinely stuck robot.
      required_movement_radius: 0.2
      # Vendor uses 10.0. We keep 20.0 (our navigator's no_progress_sec):
      # waiting out a person who steps aside is better than aborting the goal.
      movement_time_allowance: 20.0

    general_goal_checker:
      plugin: "nav2_controller::SimpleGoalChecker"
      xy_goal_tolerance: 0.25           # vendor general_goal_checker
      yaw_goal_tolerance: 0.25          # vendor general_goal_checker
