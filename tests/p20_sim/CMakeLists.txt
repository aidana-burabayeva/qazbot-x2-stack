cmake_minimum_required(VERSION 3.16)
project(p20_sim_tools)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(torso_odom torso_odom.cpp)
target_compile_features(torso_odom PRIVATE cxx_std_17)
target_compile_options(torso_odom PRIVATE -Wall -Wextra -Werror)
target_link_libraries(torso_odom Eigen3::Eigen)
ament_target_dependencies(torso_odom rclcpp nav_msgs tf2_msgs)

install(TARGETS torso_odom DESTINATION lib/${PROJECT_NAME})
ament_package()
